18//! Read path for vortex data files, producing the same transformed arrow
19//! record batch stream as the parquet read path.
20//!
21//! Field IDs are restored from the embedded Iceberg schema, or resolved through
22//! name mapping / positional fallback for imported files. Filters, deletes,
23//! partition constants, and metadata columns use the same Iceberg semantics as
24//! Parquet. Vortex's own layouts provide pruning; Parquet page indexes, INT96
25//! coercion and modular encryption do not apply here. Byte splits are mapped
26//! proportionally to row ranges using the file size and exact row count.

76/// Reads a [`FileScanTask`] pointing at a vortex data file and returns a
77/// stream of arrow record batches matching the task's projected schema.
78pub(super) async fn read_vortex_task(
79    task: FileScanTask,
80    file_io: &FileIO,
81    batch_size: Option<usize>,
82    delete_file_loader: &CachingDeleteFileLoader,
83    bytes_read_counter: Arc<AtomicU64>,
84) -> Result<ArrowRecordBatchStream> {
85    if batch_size == Some(0) {
86        return Err(Error::new(
87            ErrorKind::DataInvalid,
88            "Vortex output batch size must be greater than zero",
89        ));
90    }
91    // Start loading the delete files concurrently with opening the data file.
92    let delete_filter_rx = delete_file_loader.load_deletes(task.deletes(), task.schema_ref());
93
94    // The session captures the current tokio runtime handle at construction
95    // time, so it is created here, inside the runtime driving the scan.
96    let session = VortexSession::default();
97
98    let (vortex_file, file_size) = open_vortex_file(
99        task.data_file_path(),
100        task.file_size_in_bytes(),
101        task.key_metadata(),
102        file_io,
103        bytes_read_counter,
104        &session,
105    )
106    .await?;
107    let row_range = if task.start() == 0 && task.length() == 0 {
108        0..vortex_file.row_count()
109    } else {
110        byte_range_to_rows(
111            task.start(),
112            task.length(),
113            file_size,
114            vortex_file.row_count(),
115        )
116    };
117    if row_range.is_empty() {
118        return Ok(Box::pin(futures::stream::empty()));
119    }
120    let file_schema = match vortex_file.metadata_segment(VORTEX_SCHEMA_KEY) {
121        Some(bytes) => Arc::new(serde_json::from_slice::<Schema>(bytes.as_slice())?),
122        None => {
123            let arrow_schema = Arc::new(
124                session
125                    .arrow()
126                    .to_arrow_schema(vortex_file.dtype())
127                    .map_err(to_iceberg_error)?,
128            );
129            let arrow_schema = if let Some(mapping) = task.name_mapping() {
130                super::apply_name_mapping_to_arrow_schema(arrow_schema, mapping)?
131            } else {
132                super::add_fallback_field_ids_to_arrow_schema(&arrow_schema)
133            };
134            Arc::new(Schema::try_from(arrow_schema.as_ref())?)
135        }
136    };
137    let project_pos = task.project_field_ids().contains(&RESERVED_FIELD_ID_POS);
138    let project_row_id = task.project_field_ids().contains(&RESERVED_FIELD_ID_ROW_ID);
139    let need_pos = project_pos || (project_row_id && task.first_row_id().is_some());
140    let projected_names: Vec<_> = file_schema
141        .as_struct()
142        .fields()
143        .iter()
144        .filter(|field| {
145            task.project_field_ids().iter().any(|id| {
146                *id == field.id
147                    || file_schema
148                        .name_by_field_id(*id)
149                        .is_some_and(|name| name.starts_with(&format!("{}.", field.name)))
150            })
151        })
152        .filter(|field| {
153            !is_metadata_field(field.id)
154                || (field.id == RESERVED_FIELD_ID_ROW_ID && task.first_row_id().is_some())
155                || (field.id == RESERVED_FIELD_ID_LAST_UPDATED_SEQUENCE_NUMBER
156                    && task.first_row_id().is_some())
157        })
158        .map(|f| f.name.clone())
159        .collect();
160
161    let delete_filter = delete_filter_rx.await.unwrap()?;
162    let delete_predicate = delete_filter.build_equality_delete_predicate(&task).await?;
163
164    // In addition to the optional predicate supplied in the `FileScanTask`,
165    // we also have an optional predicate resulting from equality delete files.
166    // If both are present, we logical-AND them together to form a single filter
167    // predicate that gets pushed down into the vortex scan.
168    let final_predicate = match (task.predicate(), delete_predicate) {
169        (None, None) => None,
170        (Some(predicate), None) => Some(predicate.clone()),
171        (None, Some(predicate)) => Some(predicate),
172        (Some(filter_predicate), Some(delete_predicate)) => {
173            Some(filter_predicate.clone().and(delete_predicate))
174        }
175    };
176    let filter = final_predicate
177        .as_ref()
178        .map(|predicate| {
179            convert_predicate_to_vortex(predicate, &file_schema, vortex_file.dtype())?
180                .bind(vortex_file.dtype())
181                .map_err(to_iceberg_error)
182        })
183        .transpose()?;
184
185    let mut projection = select(
186        projected_names
187            .iter()
188            .map(String::as_str)
189            .collect::<Vec<_>>(),
190        root(),
191    );
192    // Keep the virtual position separate from user columns with the same name.
193    let mut position_name = RESERVED_COL_NAME_POS.to_string();
194    while projected_names.contains(&position_name) {
195        position_name.push('_');
196    }
197    if need_pos {
198        projection = vortex::expr::merge([
199            projection,
200            vortex::expr::pack(
201                [(
202                    position_name.as_str(),
203                    vortex::expr::cast(
204                        vortex::layout::layouts::row_idx::row_idx(),
205                        DType::Primitive(PType::I64, Nullability::NonNullable),
206                    ),
207                )],
208                Nullability::NonNullable,
209            ),
210        ]);
211    }
212    let mut scan_builder = vortex_file
213        .scan()
214        .map_err(to_iceberg_error)?
215        .with_projection(
216            projection
217                .bind(vortex_file.dtype())
218                .map_err(to_iceberg_error)?,
219        )
220        .with_some_filter(filter)
221        .with_row_range(row_range);
222    // Positional deletes are applied by excluding the deleted row ordinals
223    // from the scan. Vortex selections compose with the filter expression.
224    if let Some(positional_deletes) = delete_filter.get_delete_vector(&task) {
225        scan_builder = scan_builder.with_selection(Selection::ExcludeRoaring(
226            positional_deletes.lock().unwrap().iter().collect(),
227        ));
228    }
229
230    let array_stream = scan_builder.into_array_stream().map_err(to_iceberg_error)?;
231
232    // Mirror the parquet pipeline: adapt file batches to the task schema.
233    let mut record_batch_transformer_builder =
234        RecordBatchTransformerBuilder::new(task.schema_ref(), task.project_field_ids());
235    if task.project_field_ids().contains(&RESERVED_FIELD_ID_FILE) {
236        record_batch_transformer_builder = record_batch_transformer_builder.with_constant(
237            RESERVED_FIELD_ID_FILE,
238            Datum::string(task.data_file_path().to_string()),
239        );
240    }
241    if let (Some(partition_spec), Some(partition_data)) =
242        (task.partition_spec().cloned(), task.partition().cloned())
243    {
244        record_batch_transformer_builder =
245            record_batch_transformer_builder.with_partition(partition_spec, partition_data)?;
246    }
247    if task
248        .project_field_ids()
249        .contains(&RESERVED_FIELD_ID_SPEC_ID)
250    {
251        let spec = task
252            .partition_spec()
253            .ok_or_else(|| Error::new(ErrorKind::Unexpected, "Partition spec is missing"))?;
254        record_batch_transformer_builder = record_batch_transformer_builder
255            .with_constant(RESERVED_FIELD_ID_SPEC_ID, Datum::int(spec.spec_id()));
256    }
257    if project_pos {
258        record_batch_transformer_builder =
259            record_batch_transformer_builder.with_virtual_field(RESERVED_FIELD_ID_POS);
260    }
261    if project_row_id {
262        record_batch_transformer_builder =
263            record_batch_transformer_builder.with_virtual_field(RESERVED_FIELD_ID_ROW_ID);
264    }
265    if task
266        .project_field_ids()
267        .contains(&RESERVED_FIELD_ID_LAST_UPDATED_SEQUENCE_NUMBER)
268    {
269        record_batch_transformer_builder = match (task.first_row_id(), task.data_sequence_number())
270        {
271            (None, _) => record_batch_transformer_builder
272                .with_null_metadata_column(RESERVED_FIELD_ID_LAST_UPDATED_SEQUENCE_NUMBER)?,
273            (Some(_), Some(seq)) => {
274                if file_schema
275                    .field_by_id(RESERVED_FIELD_ID_LAST_UPDATED_SEQUENCE_NUMBER)
276                    .is_some()
277                {
278                    record_batch_transformer_builder.with_coalesced_last_updated_seq_column(
279                        RESERVED_FIELD_ID_LAST_UPDATED_SEQUENCE_NUMBER,
280                        Datum::long(seq),
281                    )
282                } else {
283                    record_batch_transformer_builder.with_constant(
284                        RESERVED_FIELD_ID_LAST_UPDATED_SEQUENCE_NUMBER,
285                        Datum::long(seq),
286                    )
287                }
288            }
289            (Some(_), None) => {
290                return Err(Error::new(
291                    ErrorKind::DataInvalid,
292                    format!(
293                        "Data file {} has a first_row_id but no data sequence number",
294                        task.data_file_path()
295                    ),
296                ));
297            }
298        };
299    }
300    if task
301        .project_field_ids()
302        .contains(&RESERVED_FIELD_ID_PARTITION)
303        && let Some(unified_type) = task.unified_partition_type()
304    {
305        let (spec, data) = match (task.partition_spec(), task.partition()) {
306            (Some(spec), Some(data)) => (Arc::clone(spec), data.clone()),
307            _ if unified_type.fields().is_empty() => (
308                Arc::new(crate::spec::PartitionSpec::unpartition_spec()),
309                crate::spec::Struct::empty(),
310            ),
311            _ => {
312                return Err(Error::new(
313                    ErrorKind::Unexpected,
314                    "cannot build _partition column: unified partition type has fields but the scan task is missing its partition spec or data",
315                ));
316            }
317        };
318        record_batch_transformer_builder =
319            record_batch_transformer_builder.with_partition_constant(
320                crate::arrow::build_partition_constant(unified_type, &spec, &data)?,
321            );
322    }
323    let mut record_batch_transformer = record_batch_transformer_builder.build();
324
325    let arrow_field = session
326        .arrow()
327        .to_arrow_field("", array_stream.dtype())
328        .map_err(to_iceberg_error)?;
329    let mut execution_ctx = session.create_execution_ctx();
330    let first_row_id = task.first_row_id();
331    let task_schema = task.schema_ref();
332
333    let record_batch_stream = array_stream.map(move |chunk| {
334        let chunk = chunk.map_err(to_iceberg_error)?;
335        let arrow_array = session
336            .arrow()
337            .execute_arrow(chunk, Some(&arrow_field), &mut execution_ctx)
338            .map_err(to_iceberg_error)?;
339        let batch = RecordBatch::from(arrow_array.as_struct());
340        let mut batch = attach_field_id_metadata(batch, &file_schema)?;
341        if need_pos {
342            let schema = batch.schema();
343            let fields: Vec<_> = schema
344                .fields()
345                .iter()
346                .map(|f| {
347                    if f.name() == &position_name {
348                        Arc::new(
349                            ArrowField::new(RESERVED_COL_NAME_POS, DataType::Int64, false)
350                                .with_metadata(std::collections::HashMap::from([(
351                                    PARQUET_FIELD_ID_META_KEY.to_string(),
352                                    RESERVED_FIELD_ID_POS.to_string(),
353                                )])),
354                        )
355                    } else {
356                        f.clone()
357                    }
358                })
359                .collect();
360            batch = RecordBatch::try_new(
361                Arc::new(arrow_schema::Schema::new(fields)),
362                batch.columns().to_vec(),
363            )?;
364        }
365        if project_row_id {
366            batch = super::row_lineage::synthesize_row_id_column(batch, first_row_id)?;
367        }
368        record_batch_transformer.process_record_batch(crate::arrow::evolve_nested_fields(
369            batch,
370            &task_schema,
371            &session,
372        )?)
373    });
374
375    if let Some(batch_size) = batch_size {
376        let record_batch_stream = record_batch_stream
377            .map_ok(move |batch| {
378                let rows = batch.num_rows();
379                let slices = rows.div_ceil(batch_size).max(1);
380                output_batches((0..slices).map(move |index| {
381                    let offset = index * batch_size;
382                    Ok::<_, Error>(batch.slice(offset, (rows - offset).min(batch_size)))
383                }))
384            })
385            .try_flatten();
386        Ok(Box::pin(record_batch_stream) as ArrowRecordBatchStream)
387    } else {
388        Ok(Box::pin(record_batch_stream) as ArrowRecordBatchStream)
389    }
390}

392/// Open Vortex files through the same counted FileIO used by data and delete scans.
393async fn open_vortex_file(
394    path: &str,
395    file_size: u64,
396    key_metadata: Option<&[u8]>,
397    file_io: &FileIO,
398    bytes_read_counter: Arc<AtomicU64>,
399    session: &VortexSession,
400) -> Result<(vortex::file::VortexFile, u64)> {
401    if key_metadata.is_some() {
402        return Err(Error::new(
403            ErrorKind::FeatureUnsupported,
404            "Encrypted Vortex files are not supported",
405        ));
406    }
407    let input_file = file_io.new_input(path)?;
408    let file_size = if file_size > 0 {
409        file_size
410    } else {
411        input_file.metadata().await?.size
412    };
413    let reader = CountingFileRead::new(input_file.reader().await?, bytes_read_counter);
414    let source = Arc::new(FileReadVortexSource {
415        reader: Arc::new(reader),
416        size: file_size,
417        uri: Arc::from(path),
418    });
419    let file = session
420        .open_options()
421        .with_file_size(file_size)
422        .with_include_metadata(true)
423        .open(source)
424        .await
425        .map_err(to_iceberg_error)?;
426    Ok((file, file_size))
427}

429/// Read unfiltered batches for delete files without recursively loading deletes.
430pub(crate) async fn vortex_to_batch_stream(
431    path: &str,
432    file_size: u64,
433    key_metadata: Option<&[u8]>,
434    file_io: &FileIO,
435    bytes_read_counter: Arc<AtomicU64>,
436) -> Result<ArrowRecordBatchStream> {
437    let session = VortexSession::default();
438    let (file, _) = open_vortex_file(
439        path,
440        file_size,
441        key_metadata,
442        file_io,
443        bytes_read_counter,
444        &session,
445    )
446    .await?;
447    let schema = file
448        .metadata_segment(VORTEX_SCHEMA_KEY)
449        .map(|bytes| serde_json::from_slice::<Schema>(bytes.as_slice()))
450        .transpose()?;
451    let stream = file
452        .scan()
453        .map_err(to_iceberg_error)?
454        .into_array_stream()
455        .map_err(to_iceberg_error)?;
456    let target_schema = schema
457        .as_ref()
458        .map(arrow_schema::Schema::try_from)
459        .transpose()?
460        .map(Arc::new);
461    let field = session
462        .arrow()
463        .to_arrow_field("", stream.dtype())
464        .map_err(to_iceberg_error)?;
465    let mut ctx = session.create_execution_ctx();
466    Ok(Box::pin(stream.map(move |chunk| {
467        let array = session
468            .arrow()
469            .execute_arrow(chunk.map_err(to_iceberg_error)?, Some(&field), &mut ctx)
470            .map_err(to_iceberg_error)?;
471        let batch = RecordBatch::from(array.as_struct());
472        match &schema {
473            Some(schema) => {
474                let batch = attach_field_id_metadata(batch, schema)?;
475                let batch = crate::arrow::evolve_nested_fields(batch, schema, &session)?;
476                let target_schema = target_schema.as_ref().expect("schema converted above");
477                let columns = batch
478                    .columns()
479                    .iter()
480                    .zip(target_schema.fields())
481                    .map(|(array, field)| arrow_cast::cast(array, field.data_type()))
482                    .collect::<std::result::Result<Vec<_>, _>>()?;
483                Ok(RecordBatch::try_new_with_options(
484                    target_schema.clone(),
485                    columns,
486                    &arrow_array::RecordBatchOptions::new()
487                        .with_row_count(Some(batch.num_rows()))
488                        .with_match_field_names(false),
489                )?)
490            }
491            None => Ok(batch),
492        }
493    })))
494}

496/// A [`VortexReadAt`] source backed by iceberg's [`FileRead`].
497struct FileReadVortexSource {
498    reader: Arc<dyn FileRead>,
499    size: u64,
500    uri: Arc<str>,
501}
502
503impl VortexReadAt for FileReadVortexSource {
504    fn uri(&self) -> Option<&Arc<str>> {
505        Some(&self.uri)
506    }
507
508    fn concurrency(&self) -> usize {
509        DEFAULT_RANGE_FETCH_CONCURRENCY
510    }
511
512    fn size(&self) -> BoxFuture<'static, VortexResult<u64>> {
513        futures::future::ready(Ok(self.size)).boxed()
514    }
515
516    fn read_at(
517        &self,
518        offset: u64,
519        length: usize,
520        alignment: Alignment,
521    ) -> BoxFuture<'static, VortexResult<BufferHandle>> {
522        let reader = Arc::clone(&self.reader);
523        async move {
524            let bytes = reader
525                .read(offset..offset + length as u64)
526                .await
527                .map_err(|err| vortex_err!("Failed to read from iceberg storage: {err}"))?;
528            if bytes.len() != length {
529                return Err(vortex_err!(
530                    "Short read from iceberg storage: expected {length} bytes, got {}",
531                    bytes.len()
532                ));
533            }
534            Ok(BufferHandle::new_host(
535                ByteBuffer::from(bytes).aligned(alignment),
536            ))
537        }
538        .boxed()
539    }
540}

542/// Converts an iceberg [`BoundPredicate`] into a vortex filter [`Expression`].
543///
544/// Vortex applies filters exactly (not just for pruning), matching the row
545/// level filtering guarantees of the parquet read path. Like the parquet read
546/// path, predicate leaves referencing columns that do not exist in the file
547/// evaluate to "keep all rows".
548fn convert_predicate_to_vortex(
549    predicate: &BoundPredicate,
550    schema: &Schema,
551    file_dtype: &DType,
552) -> Result<Expression> {
553    match predicate {
554        BoundPredicate::AlwaysTrue => Ok(lit(true)),
555        BoundPredicate::AlwaysFalse => Ok(lit(false)),
556        BoundPredicate::And(expr) => {
557            let [left, right] = expr.inputs();
558            Ok(and(
559                convert_predicate_to_vortex(left, schema, file_dtype)?,
560                convert_predicate_to_vortex(right, schema, file_dtype)?,
561            ))
562        }
563        BoundPredicate::Or(expr) => {
564            let [left, right] = expr.inputs();
565            Ok(or(
566                convert_predicate_to_vortex(left, schema, file_dtype)?,
567                convert_predicate_to_vortex(right, schema, file_dtype)?,
568            ))
569        }
570        BoundPredicate::Not(expr) => {
571            let [inner] = expr.inputs();
572            Ok(not(convert_predicate_to_vortex(inner, schema, file_dtype)?))
573        }
574        BoundPredicate::Unary(expr) => {
575            let Some(column) = column_expr(expr.term(), schema, file_dtype)? else {
576                return Ok(lit(true));
577            };
578            match expr.op() {
579                PredicateOperator::IsNull => Ok(is_null(column.expr)),
580                PredicateOperator::NotNull => Ok(is_not_null(column.expr)),
581                PredicateOperator::IsNan => nan_predicate(column.expr, expr.term().field(), true),
582                PredicateOperator::NotNan => nan_predicate(column.expr, expr.term().field(), false),
583                op => Err(unsupported_predicate(op)),
584            }
585        }
586        BoundPredicate::Binary(expr) => {
587            let Some(column) = column_expr(expr.term(), schema, file_dtype)? else {
588                return Ok(lit(true));
589            };
590            match expr.op() {
591                PredicateOperator::StartsWith => {
592                    starts_with_predicate(column.expr, expr.literal(), false)
593                }
594                PredicateOperator::NotStartsWith => {
595                    starts_with_predicate(column.expr, expr.literal(), true)
596                }
597                op => {
598                    let literal = lit(datum_to_vortex_scalar(expr.literal(), &column.dtype)?);
599                    match op {
600                        PredicateOperator::LessThan => Ok(lt(column.expr, literal)),
601                        PredicateOperator::LessThanOrEq => Ok(lt_eq(column.expr, literal)),
602                        PredicateOperator::GreaterThan => Ok(gt(column.expr, literal)),
603                        PredicateOperator::GreaterThanOrEq => Ok(gt_eq(column.expr, literal)),
604                        PredicateOperator::Eq => Ok(eq(column.expr, literal)),
605                        PredicateOperator::NotEq => Ok(not_eq(column.expr, literal)),
606                        op => Err(unsupported_predicate(op)),
607                    }
608                }
609            }
610        }
611        BoundPredicate::Set(expr) => {
612            let Some(column) = column_expr(expr.term(), schema, file_dtype)? else {
613                return Ok(lit(true));
614            };
615            let matches = expr
616                .literals()
617                .iter()
618                .map(|datum| {
619                    Ok(eq(
620                        column.expr.clone(),
621                        lit(datum_to_vortex_scalar(datum, &column.dtype)?),
622                    ))
623                })
624                .collect::<Result<Vec<_>>>()?;
625            match expr.op() {
626                PredicateOperator::In => Ok(or_collect(matches).unwrap_or_else(|| lit(false))),
627                PredicateOperator::NotIn => Ok(match or_collect(matches) {
628                    Some(any_match) => not(any_match),
629                    None => lit(true),
630                }),
631                op => Err(unsupported_predicate(op)),
632            }
633        }
634    }
635}

640        format!("Predicate operator {op} is not yet supported for vortex data files"),

644/// A column reference resolved against the file: the vortex expression that
645/// selects it, plus its dtype in the file.
646struct FileColumn {
647    expr: Expression,
648    dtype: DType,
649}
650
651/// Resolves the field referenced by a bound predicate term to a vortex column
652/// expression. Nested struct fields become chains of `get_item`.
653///
654/// Returns `None` when the column (or any struct along its path) does not
655/// exist in the file, in which case the predicate leaf must evaluate to
656/// "keep all rows", mirroring the parquet read path.
657fn column_expr(
658    reference: &BoundReference,
659    schema: &Schema,
660    file_dtype: &DType,
661) -> Result<Option<FileColumn>> {
662    // Walk the accessor positions through the table schema to build the
663    // column's name path.
664    let mut names: Vec<String> = Vec::new();
665    let mut current_struct = Some(schema.as_struct());
666    let Some(file_accessor) = schema.accessor_by_field_id(reference.field().id) else {
667        return Ok(None);
668    };
669    let mut accessor = Some(file_accessor.as_ref());
670    while let Some(acc) = accessor {
671        let field = current_struct
672            .and_then(|struct_type| struct_type.fields().get(acc.position()))
673            .ok_or_else(|| {
674                Error::new(
675                    ErrorKind::Unexpected,
676                    format!(
677                        "Bound reference {} does not resolve within the table schema",
678                        reference.field().name
679                    ),
680                )
681            })?;
682        names.push(field.name.clone());
683        accessor = acc.inner();
684        current_struct = match field.field_type.as_ref() {
685            Type::Struct(struct_type) => Some(struct_type),
686            _ => None,
687        };
688    }
689
690    // Resolve the same path in the file's dtype; vortex files are matched by
691    // name.
692    let mut dtype = file_dtype.clone();
693    for name in &names {
694        let Some(field_dtype) = dtype
695            .as_struct_fields_opt()
696            .and_then(|fields| fields.field(name))
697        else {
698            return Ok(None);
699        };
700        dtype = field_dtype;
701    }
702
703    let mut expr = names
704        .iter()
705        .fold(root(), |child, name| get_item(name.as_str(), child));
706    // Promote the column instead of narrowing the literal: out-of-range literals
707    // must remain valid filters after int/float/decimal schema evolution.
708    let promoted = match (&dtype, reference.field().field_type.as_ref()) {
709        (DType::Primitive(PType::I32, nullability), Type::Primitive(PrimitiveType::Long)) => {
710            Some(DType::Primitive(PType::I64, *nullability))
711        }
712        (DType::Primitive(PType::F32, nullability), Type::Primitive(PrimitiveType::Double)) => {
713            Some(DType::Primitive(PType::F64, *nullability))
714        }
715        (
716            DType::Decimal(_, nullability),
717            Type::Primitive(PrimitiveType::Decimal { precision, scale }),
718        ) => Some(DType::Decimal(
719            DecimalDType::new(u8::try_from(*precision)?, i8::try_from(*scale)?),
720            *nullability,
721        )),
722        _ => None,
723    };
724    if let Some(promoted) = promoted
725        && promoted != dtype
726    {
727        expr = vortex::expr::cast(expr, promoted.clone());
728        dtype = promoted;
729    }
730    Ok(Some(FileColumn { expr, dtype }))
731}

733/// Builds a NaN check for a float column.
734///
735/// Vortex compares floats using IEEE-754 total ordering, in which every NaN
736/// bit pattern sorts either above `+inf` (positive NaNs) or below `-inf`
737/// (negative NaNs), so NaN checks are expressed as comparisons against the
738/// infinities. Null values compare to null and are excluded by the filter,
739/// matching the three valued logic of the parquet read path.
740fn nan_predicate(column: Expression, field: &NestedField, is_nan: bool) -> Result<Expression> {
741    let (pos_inf, neg_inf): (Scalar, Scalar) = match field.field_type.as_ref() {
742        Type::Primitive(PrimitiveType::Float) => (f32::INFINITY.into(), f32::NEG_INFINITY.into()),
743        Type::Primitive(PrimitiveType::Double) => (f64::INFINITY.into(), f64::NEG_INFINITY.into()),
744        ty => {
745            return Err(Error::new(
746                ErrorKind::Unexpected,
747                format!(
748                    "NaN predicate on non-float field {} of type {ty}",
749                    field.name
750                ),
751            ));
752        }
753    };
754    Ok(if is_nan {
755        or(gt(column.clone(), lit(pos_inf)), lt(column, lit(neg_inf)))
756    } else {
757        and(
758            lt_eq(column.clone(), lit(pos_inf)),
759            gt_eq(column, lit(neg_inf)),
760        )
761    })
762}

764/// Builds a `LIKE` pattern matching values that start with the literal prefix,
765/// escaping any SQL LIKE wildcards contained in it.
766fn starts_with_predicate(column: Expression, prefix: &Datum, negated: bool) -> Result<Expression> {
767    let PrimitiveLiteral::String(prefix) = prefix.literal() else {
768        return Err(Error::new(
769            ErrorKind::Unexpected,
770            "STARTS WITH predicates require a string literal",
771        ));
772    };
773    let escaped = prefix
774        .replace('\\', "\\\\")
775        .replace('%', "\\%")
776        .replace('_', "\\_");
777    let pattern = format!("{escaped}%");
778    Ok(if negated {
779        not_like(column, lit(pattern))
780    } else {
781        like(column, lit(pattern))
782    })
783}

785fn datum_to_vortex_scalar(datum: &Datum, file_dtype: &DType) -> Result<Scalar> {
786    match (datum.data_type(), datum.literal()) {
787        (PrimitiveType::Boolean, PrimitiveLiteral::Boolean(v)) => Ok((*v).into()),
788        (PrimitiveType::Int, PrimitiveLiteral::Int(v)) => Ok((*v).into()),
789        (PrimitiveType::Long, PrimitiveLiteral::Long(v)) => Ok((*v).into()),
790        (PrimitiveType::Float, PrimitiveLiteral::Float(v)) => Ok(v.0.into()),
791        (PrimitiveType::Double, PrimitiveLiteral::Double(v)) => Ok(v.0.into()),
792        (PrimitiveType::String, PrimitiveLiteral::String(v)) => Ok(v.as_str().into()),
793        (PrimitiveType::Binary, PrimitiveLiteral::Binary(v)) => Ok(v.as_slice().into()),
794        (PrimitiveType::Uuid, PrimitiveLiteral::UInt128(value)) => {
795            let DType::Extension(ext) = file_dtype else {
796                return Err(Error::new(
797                    ErrorKind::FeatureUnsupported,
798                    format!("Cannot compare UUID to Vortex column of type {file_dtype}"),
799                ));
800            };
801            if !ext.is::<Uuid>() {
802                return Err(Error::new(
803                    ErrorKind::FeatureUnsupported,
804                    format!("Cannot compare UUID to Vortex extension {ext}"),
805                ));
806            }
807            let storage = Scalar::fixed_size_list(
808                DType::Primitive(PType::U8, Nullability::NonNullable),
809                value.to_be_bytes().into_iter().map(Scalar::from).collect(),
810                ext.storage_dtype().nullability(),
811            );
812            Ok(Scalar::extension_ref(ext.clone(), storage))
813        }
814        (PrimitiveType::Decimal { precision, scale }, PrimitiveLiteral::Int128(v)) => {
815            let precision = u8::try_from(*precision).map_err(|err| {
816                Error::new(
817                    ErrorKind::DataInvalid,
818                    format!("Decimal precision {precision} out of range"),
819                )
820                .with_source(err)
821            })?;
822            let scale = i8::try_from(*scale).map_err(|err| {
823                Error::new(
824                    ErrorKind::DataInvalid,
825                    format!("Decimal scale {scale} out of range"),
826                )
827                .with_source(err)
828            })?;
829            Ok(Scalar::decimal(
830                DecimalValue::I128(*v),
831                DecimalDType::new(precision, scale),
832                Nullability::NonNullable,
833            ))
834        }
835        (PrimitiveType::Date, PrimitiveLiteral::Int(days)) => {
836            temporal_scalar(file_dtype, i64::from(*days), TimeUnit::Days)
837        }
838        (PrimitiveType::Time, PrimitiveLiteral::Long(micros)) => {
839            temporal_scalar(file_dtype, *micros, TimeUnit::Microseconds)
840        }
841        (PrimitiveType::Timestamp | PrimitiveType::Timestamptz, PrimitiveLiteral::Long(micros)) => {
842            temporal_scalar(file_dtype, *micros, TimeUnit::Microseconds)
843        }
844        (
845            PrimitiveType::TimestampNs | PrimitiveType::TimestamptzNs,
846            PrimitiveLiteral::Long(nanos),
847        ) => temporal_scalar(file_dtype, *nanos, TimeUnit::Nanoseconds),
848        (ty, _) => Err(Error::new(
849            ErrorKind::FeatureUnsupported,
850            format!("Predicate literals of type {ty} are not yet supported for vortex data files"),
851        )),
852    }
853}

855/// Map byte boundaries using average row size, as in Iceberg Java. Using the
856/// same floor operation at each boundary makes adjacent splits disjoint and
857/// exhaustive. Widening the product avoids overflow and floating-point rounding.
858fn byte_range_to_rows(start: u64, length: u64, file_size: u64, rows: u64) -> std::ops::Range<u64> {
859    let row_at = |offset: u64| {
860        if offset >= file_size {
861            rows
862        } else {
863            ((u128::from(offset) * u128::from(rows)) / u128::from(file_size)) as u64
864        }
865    };
866    row_at(start)..row_at(start.saturating_add(length))
867}

869/// Builds a literal scalar for comparing against a temporal column.
870///
871/// When the file stores the column as a vortex temporal extension type, the
872/// literal is converted to the column's exact extension dtype (unit and
873/// timezone), since vortex only compares extension values against matching
874/// extension dtypes. Plain primitive columns are compared against the raw
875/// iceberg value.
876fn temporal_scalar(file_dtype: &DType, value: i64, value_unit: TimeUnit) -> Result<Scalar> {
877    match file_dtype {
878        DType::Extension(ext) => {
879            let Some(metadata) = AnyTemporal::try_match(ext) else {
880                return Err(Error::new(
881                    ErrorKind::FeatureUnsupported,
882                    format!(
883                        "Cannot compare a temporal predicate literal to a column of type {file_dtype}"
884                    ),
885                ));
886            };
887            let converted = convert_temporal_value(value, value_unit, metadata.time_unit())?;
888            let nullability = ext.storage_dtype().nullability();
889            let storage = match ext.storage_dtype() {
890                DType::Primitive(PType::I32, _) => Scalar::primitive(
891                    i32::try_from(converted).map_err(|err| {
892                        Error::new(
893                            ErrorKind::DataInvalid,
894                            format!("Temporal value {converted} overflows the column storage type"),
895                        )
896                        .with_source(err)
897                    })?,
898                    nullability,
899                ),
900                DType::Primitive(PType::I64, _) => Scalar::primitive(converted, nullability),
901                dtype => {
902                    return Err(Error::new(
903                        ErrorKind::FeatureUnsupported,
904                        format!("Unsupported temporal storage type {dtype}"),
905                    ));
906                }
907            };
908            Ok(Scalar::extension_ref(ext.clone(), storage))
909        }
910        DType::Primitive(PType::I32, _) => Ok(Scalar::primitive(
911            i32::try_from(value).map_err(|err| {
912                Error::new(
913                    ErrorKind::DataInvalid,
914                    format!("Temporal value {value} overflows the column storage type"),
915                )
916                .with_source(err)
917            })?,
918            Nullability::NonNullable,
919        )),
920        DType::Primitive(PType::I64, _) => Ok(Scalar::primitive(value, Nullability::NonNullable)),
921        dtype => Err(Error::new(
922            ErrorKind::FeatureUnsupported,
923            format!("Cannot compare a temporal predicate literal to a column of type {dtype}"),
924        )),
925    }
926}
