63/// Reads a [`FileScanTask`] pointing at a vortex data file and returns a
64/// stream of arrow record batches matching the task's projected schema.
65pub(super) async fn read_vortex_task(
66    task: FileScanTask,
67    file_io: &FileIO,
68    batch_size: Option<usize>,
69    delete_file_loader: &CachingDeleteFileLoader,
70    bytes_read_counter: Arc<AtomicU64>,
71) -> Result<ArrowRecordBatchStream> {
72    // Start loading the delete files concurrently with opening the data file.
73    let delete_filter_rx = delete_file_loader.load_deletes(&task.deletes, Arc::clone(&task.schema));
74
75    // The session captures the current tokio runtime handle at construction
76    // time, so it is created here, inside the runtime driving the scan.
77    let session = VortexSession::default();
78
79    // Open the vortex file through iceberg's FileIO.
80    let input_file = file_io.new_input(&task.data_file_path)?;
81    let file_size = if task.file_size_in_bytes > 0 {
82        task.file_size_in_bytes
83    } else {
84        input_file.metadata().await?.size
85    };
86    let reader = CountingFileRead::new(input_file.reader().await?, bytes_read_counter);
87    let source = Arc::new(FileReadVortexSource {
88        reader: Arc::new(reader),
89        size: file_size,
90        uri: Arc::from(task.data_file_path.as_str()),
91    });
92
93    let vortex_file = session
94        .open_options()
95        .with_file_size(file_size)
96        .open(source)
97        .await
98        .map_err(to_iceberg_error)?;
99
100    // Vortex files do not record iceberg field ids, so columns are matched by
101    // name. Fields that were added to the table schema after the file was
102    // written are filled in by the RecordBatchTransformer below.
103    let file_field_names: Vec<&str> = vortex_file
104        .dtype()
105        .as_struct_fields_opt()
106        .ok_or_else(|| {
107            Error::new(
108                ErrorKind::DataInvalid,
109                "Vortex data file does not contain a struct dtype",
110            )
111        })?
112        .names()
113        .iter()
114        .map(|name| name.as_ref())
115        .collect();
116    let projected_names: Vec<String> = task
117        .project_field_ids
118        .iter()
119        .filter_map(|id| task.schema.name_by_field_id(*id))
120        .filter(|name| file_field_names.contains(name))
121        .map(|name| name.to_string())
122        .collect();
123
124    let delete_filter = delete_filter_rx.await.unwrap()?;
125    let delete_predicate = delete_filter.build_equality_delete_predicate(&task).await?;
126
127    // In addition to the optional predicate supplied in the `FileScanTask`,
128    // we also have an optional predicate resulting from equality delete files.
129    // If both are present, we logical-AND them together to form a single filter
130    // predicate that gets pushed down into the vortex scan.
131    let final_predicate = match (&task.predicate, delete_predicate) {
132        (None, None) => None,
133        (Some(predicate), None) => Some(predicate.clone()),
134        (None, Some(predicate)) => Some(predicate),
135        (Some(filter_predicate), Some(delete_predicate)) => {
136            Some(filter_predicate.clone().and(delete_predicate))
137        }
138    };
139    let filter = final_predicate
140        .as_ref()
141        .map(|predicate| convert_predicate_to_vortex(predicate, &task.schema, vortex_file.dtype()))
142        .transpose()?;
143
144    let mut scan_builder = vortex_file
145        .scan()
146        .map_err(to_iceberg_error)?
147        .with_projection(select(
148            projected_names
149                .iter()
150                .map(|s| s.as_str())
151                .collect::<Vec<_>>(),
152            root(),
153        ))
154        .with_some_filter(filter);
155    if let Some(batch_size) = batch_size {
156        scan_builder = scan_builder.with_split_by(SplitBy::RowCount(batch_size));
157    }
158
159    // Positional deletes are applied by excluding the deleted row ordinals
160    // from the scan. Vortex selections compose with the filter expression.
161    if let Some(positional_deletes) = delete_filter.get_delete_vector(&task) {
162        let deleted_rows: Buffer<u64> = {
163            let guard = positional_deletes.lock().unwrap();
164            guard.iter().collect()
165        };
166        scan_builder = scan_builder.with_selection(Selection::ExcludeByIndex(deleted_rows));
167    }
168
169    let array_stream = scan_builder.into_array_stream().map_err(to_iceberg_error)?;
170
171    // Mirror the parquet pipeline: adapt file batches to the task schema.
172    let mut record_batch_transformer_builder =
173        RecordBatchTransformerBuilder::new(task.schema_ref(), task.project_field_ids());
174    if task.project_field_ids().contains(&RESERVED_FIELD_ID_FILE) {
175        record_batch_transformer_builder = record_batch_transformer_builder.with_constant(
176            RESERVED_FIELD_ID_FILE,
177            Datum::string(task.data_file_path.clone()),
178        );
179    }
180    if let (Some(partition_spec), Some(partition_data)) =
181        (task.partition_spec.clone(), task.partition.clone())
182    {
183        record_batch_transformer_builder =
184            record_batch_transformer_builder.with_partition(partition_spec, partition_data)?;
185    }
186    let mut record_batch_transformer = record_batch_transformer_builder.build();
187
188    let arrow_field = session
189        .arrow()
190        .to_arrow_field("", array_stream.dtype())
191        .map_err(to_iceberg_error)?;
192    let mut execution_ctx = session.create_execution_ctx();
193    let task_schema = task.schema_ref();
194
195    let record_batch_stream = array_stream.map(move |chunk| {
196        let chunk = chunk.map_err(to_iceberg_error)?;
197        let arrow_array = session
198            .arrow()
199            .execute_arrow(chunk, Some(&arrow_field), &mut execution_ctx)
200            .map_err(to_iceberg_error)?;
201        let batch = RecordBatch::from(arrow_array.as_struct());
202        let batch = attach_field_id_metadata(batch, &task_schema)?;
203        record_batch_transformer.process_record_batch(batch)
204    });
205
206    Ok(Box::pin(record_batch_stream) as ArrowRecordBatchStream)
207}

209/// Tags every top-level column of `batch` that exists in `schema` with its
210/// iceberg field id, so that the [`RecordBatchTransformer`] can match columns
211/// by id like it does for parquet files.
212fn attach_field_id_metadata(batch: RecordBatch, schema: &Schema) -> Result<RecordBatch> {
213    let fields: Vec<ArrowField> = batch
214        .schema()
215        .fields()
216        .iter()
217        .map(|field| {
218            let field = field.as_ref().clone();
219            match schema.field_id_by_name(field.name()) {
220                Some(field_id) => {
221                    let mut metadata = field.metadata().clone();
222                    metadata.insert(PARQUET_FIELD_ID_META_KEY.to_string(), field_id.to_string());
223                    field.with_metadata(metadata)
224                }
225                None => field,
226            }
227        })
228        .collect();
229    let schema = Arc::new(ArrowSchema::new(fields));
230    RecordBatch::try_new(schema, batch.columns().to_vec()).map_err(|err| {
231        Error::new(
232            ErrorKind::Unexpected,
233            "Failed to attach field id metadata to record batch",
234        )
235        .with_source(err)
236    })
237}

239/// A [`VortexReadAt`] source backed by iceberg's [`FileRead`].
240struct FileReadVortexSource {
241    reader: Arc<dyn FileRead>,
242    size: u64,
243    uri: Arc<str>,
244}
245
246impl VortexReadAt for FileReadVortexSource {
247    fn uri(&self) -> Option<&Arc<str>> {
248        Some(&self.uri)
249    }
250
251    fn concurrency(&self) -> usize {
252        DEFAULT_RANGE_FETCH_CONCURRENCY
253    }
254
255    fn size(&self) -> BoxFuture<'static, VortexResult<u64>> {
256        futures::future::ready(Ok(self.size)).boxed()
257    }
258
259    fn read_at(
260        &self,
261        offset: u64,
262        length: usize,
263        alignment: Alignment,
264    ) -> BoxFuture<'static, VortexResult<BufferHandle>> {
265        let reader = Arc::clone(&self.reader);
266        async move {
267            let bytes = reader
268                .read(offset..offset + length as u64)
269                .await
270                .map_err(|err| vortex_err!("Failed to read from iceberg storage: {err}"))?;
271            if bytes.len() != length {
272                return Err(vortex_err!(
273                    "Short read from iceberg storage: expected {length} bytes, got {}",
274                    bytes.len()
275                ));
276            }
277            Ok(BufferHandle::new_host(
278                ByteBuffer::from(bytes).aligned(alignment),
279            ))
280        }
281        .boxed()
282    }
283}

285/// Converts an iceberg [`BoundPredicate`] into a vortex filter [`Expression`].
286///
287/// Vortex applies filters exactly (not just for pruning), matching the row
288/// level filtering guarantees of the parquet read path. Like the parquet read
289/// path, predicate leaves referencing columns that do not exist in the file
290/// evaluate to "keep all rows".
291fn convert_predicate_to_vortex(
292    predicate: &BoundPredicate,
293    schema: &Schema,
294    file_dtype: &DType,
295) -> Result<Expression> {
296    match predicate {
297        BoundPredicate::AlwaysTrue => Ok(lit(true)),
298        BoundPredicate::AlwaysFalse => Ok(lit(false)),
299        BoundPredicate::And(expr) => {
300            let [left, right] = expr.inputs();
301            Ok(and(
302                convert_predicate_to_vortex(left, schema, file_dtype)?,
303                convert_predicate_to_vortex(right, schema, file_dtype)?,
304            ))
305        }
306        BoundPredicate::Or(expr) => {
307            let [left, right] = expr.inputs();
308            Ok(or(
309                convert_predicate_to_vortex(left, schema, file_dtype)?,
310                convert_predicate_to_vortex(right, schema, file_dtype)?,
311            ))
312        }
313        BoundPredicate::Not(expr) => {
314            let [inner] = expr.inputs();
315            Ok(not(convert_predicate_to_vortex(inner, schema, file_dtype)?))
316        }
317        BoundPredicate::Unary(expr) => {
318            let Some(column) = column_expr(expr.term(), schema, file_dtype)? else {
319                return Ok(lit(true));
320            };
321            match expr.op() {
322                PredicateOperator::IsNull => Ok(is_null(column.expr)),
323                PredicateOperator::NotNull => Ok(is_not_null(column.expr)),
324                PredicateOperator::IsNan => nan_predicate(column.expr, expr.term().field(), true),
325                PredicateOperator::NotNan => nan_predicate(column.expr, expr.term().field(), false),
326                op => Err(unsupported_predicate(op)),
327            }
328        }
329        BoundPredicate::Binary(expr) => {
330            let Some(column) = column_expr(expr.term(), schema, file_dtype)? else {
331                return Ok(lit(true));
332            };
333            match expr.op() {
334                PredicateOperator::StartsWith => {
335                    starts_with_predicate(column.expr, expr.literal(), false)
336                }
337                PredicateOperator::NotStartsWith => {
338                    starts_with_predicate(column.expr, expr.literal(), true)
339                }
340                op => {
341                    let literal = lit(datum_to_vortex_scalar(expr.literal(), &column.dtype)?);
342                    match op {
343                        PredicateOperator::LessThan => Ok(lt(column.expr, literal)),
344                        PredicateOperator::LessThanOrEq => Ok(lt_eq(column.expr, literal)),
345                        PredicateOperator::GreaterThan => Ok(gt(column.expr, literal)),
346                        PredicateOperator::GreaterThanOrEq => Ok(gt_eq(column.expr, literal)),
347                        PredicateOperator::Eq => Ok(eq(column.expr, literal)),
348                        PredicateOperator::NotEq => Ok(not_eq(column.expr, literal)),
349                        op => Err(unsupported_predicate(op)),
350                    }
351                }
352            }
353        }
354        BoundPredicate::Set(expr) => {
355            let Some(column) = column_expr(expr.term(), schema, file_dtype)? else {
356                return Ok(lit(true));
357            };
358            let matches = expr
359                .literals()
360                .iter()
361                .map(|datum| {
362                    Ok(eq(
363                        column.expr.clone(),
364                        lit(datum_to_vortex_scalar(datum, &column.dtype)?),
365                    ))
366                })
367                .collect::<Result<Vec<_>>>()?;
368            match expr.op() {
369                PredicateOperator::In => Ok(or_collect(matches).unwrap_or_else(|| lit(false))),
370                PredicateOperator::NotIn => Ok(match or_collect(matches) {
371                    Some(any_match) => not(any_match),
372                    None => lit(true),
373                }),
374                op => Err(unsupported_predicate(op)),
375            }
376        }
377    }
378}

380fn unsupported_predicate(op: PredicateOperator) -> Error {
381    Error::new(
382        ErrorKind::FeatureUnsupported,
383        format!("Predicate operator {op} is not yet supported for vortex data files"),
384    )
385}

387/// A column reference resolved against the file: the vortex expression that
388/// selects it, plus its dtype in the file.
389struct FileColumn {
390    expr: Expression,
391    dtype: DType,
392}
393
394/// Resolves the field referenced by a bound predicate term to a vortex column
395/// expression. Nested struct fields become chains of `get_item`.
396///
397/// Returns `None` when the column (or any struct along its path) does not
398/// exist in the file, in which case the predicate leaf must evaluate to
399/// "keep all rows", mirroring the parquet read path.
400fn column_expr(
401    reference: &BoundReference,
402    schema: &Schema,
403    file_dtype: &DType,
404) -> Result<Option<FileColumn>> {
405    // Walk the accessor positions through the table schema to build the
406    // column's name path.
407    let mut names: Vec<String> = Vec::new();
408    let mut current_struct = Some(schema.as_struct());
409    let mut accessor = Some(reference.accessor());
410    while let Some(acc) = accessor {
411        let field = current_struct
412            .and_then(|struct_type| struct_type.fields().get(acc.position()))
413            .ok_or_else(|| {
414                Error::new(
415                    ErrorKind::Unexpected,
416                    format!(
417                        "Bound reference {} does not resolve within the table schema",
418                        reference.field().name
419                    ),
420                )
421            })?;
422        names.push(field.name.clone());
423        accessor = acc.inner();
424        current_struct = match field.field_type.as_ref() {
425            Type::Struct(struct_type) => Some(struct_type),
426            _ => None,
427        };
428    }
429
430    // Resolve the same path in the file's dtype; vortex files are matched by
431    // name.
432    let mut dtype = file_dtype.clone();
433    for name in &names {
434        let Some(field_dtype) = dtype
435            .as_struct_fields_opt()
436            .and_then(|fields| fields.field(name))
437        else {
438            return Ok(None);
439        };
440        dtype = field_dtype;
441    }
442
443    let expr = names
444        .iter()
445        .fold(root(), |child, name| get_item(name.as_str(), child));
446    Ok(Some(FileColumn { expr, dtype }))
447}

449/// Builds a NaN check for a float column.
450///
451/// Vortex compares floats using IEEE-754 total ordering, in which every NaN
452/// bit pattern sorts either above `+inf` (positive NaNs) or below `-inf`
453/// (negative NaNs), so NaN checks are expressed as comparisons against the
454/// infinities. Null values compare to null and are excluded by the filter,
455/// matching the three valued logic of the parquet read path.
456fn nan_predicate(column: Expression, field: &NestedField, is_nan: bool) -> Result<Expression> {
457    let (pos_inf, neg_inf): (Scalar, Scalar) = match field.field_type.as_ref() {
458        Type::Primitive(PrimitiveType::Float) => (f32::INFINITY.into(), f32::NEG_INFINITY.into()),
459        Type::Primitive(PrimitiveType::Double) => (f64::INFINITY.into(), f64::NEG_INFINITY.into()),
460        ty => {
461            return Err(Error::new(
462                ErrorKind::Unexpected,
463                format!(
464                    "NaN predicate on non-float field {} of type {ty}",
465                    field.name
466                ),
467            ));
468        }
469    };
470    Ok(if is_nan {
471        or(gt(column.clone(), lit(pos_inf)), lt(column, lit(neg_inf)))
472    } else {
473        and(
474            lt_eq(column.clone(), lit(pos_inf)),
475            gt_eq(column, lit(neg_inf)),
476        )
477    })
478}

480/// Builds a `LIKE` pattern matching values that start with the literal prefix,
481/// escaping any SQL LIKE wildcards contained in it.
482fn starts_with_predicate(column: Expression, prefix: &Datum, negated: bool) -> Result<Expression> {
483    let PrimitiveLiteral::String(prefix) = prefix.literal() else {
484        return Err(Error::new(
485            ErrorKind::Unexpected,
486            "STARTS WITH predicates require a string literal",
487        ));
488    };
489    let escaped = prefix
490        .replace('\\', "\\\\")
491        .replace('%', "\\%")
492        .replace('_', "\\_");
493    let pattern = format!("{escaped}%");
494    Ok(if negated {
495        not_like(column, lit(pattern))
496    } else {
497        like(column, lit(pattern))
498    })
499}

501fn datum_to_vortex_scalar(datum: &Datum, file_dtype: &DType) -> Result<Scalar> {
502    match (datum.data_type(), datum.literal()) {
503        (PrimitiveType::Boolean, PrimitiveLiteral::Boolean(v)) => Ok((*v).into()),
504        (PrimitiveType::Int, PrimitiveLiteral::Int(v)) => Ok((*v).into()),
505        (PrimitiveType::Long, PrimitiveLiteral::Long(v)) => Ok((*v).into()),
506        (PrimitiveType::Float, PrimitiveLiteral::Float(v)) => Ok(v.0.into()),
507        (PrimitiveType::Double, PrimitiveLiteral::Double(v)) => Ok(v.0.into()),
508        (PrimitiveType::String, PrimitiveLiteral::String(v)) => Ok(v.as_str().into()),
509        (PrimitiveType::Binary, PrimitiveLiteral::Binary(v)) => Ok(v.as_slice().into()),
510        (PrimitiveType::Decimal { precision, scale }, PrimitiveLiteral::Int128(v)) => {
511            let precision = u8::try_from(*precision).map_err(|err| {
512                Error::new(
513                    ErrorKind::DataInvalid,
514                    format!("Decimal precision {precision} out of range"),
515                )
516                .with_source(err)
517            })?;
518            let scale = i8::try_from(*scale).map_err(|err| {
519                Error::new(
520                    ErrorKind::DataInvalid,
521                    format!("Decimal scale {scale} out of range"),
522                )
523                .with_source(err)
524            })?;
525            Ok(Scalar::decimal(
526                DecimalValue::I128(*v),
527                DecimalDType::new(precision, scale),
528                Nullability::NonNullable,
529            ))
530        }
531        (PrimitiveType::Date, PrimitiveLiteral::Int(days)) => {
532            temporal_scalar(file_dtype, i64::from(*days), TimeUnit::Days)
533        }
534        (PrimitiveType::Time, PrimitiveLiteral::Long(micros)) => {
535            temporal_scalar(file_dtype, *micros, TimeUnit::Microseconds)
536        }
537        (PrimitiveType::Timestamp | PrimitiveType::Timestamptz, PrimitiveLiteral::Long(micros)) => {
538            temporal_scalar(file_dtype, *micros, TimeUnit::Microseconds)
539        }
540        (
541            PrimitiveType::TimestampNs | PrimitiveType::TimestamptzNs,
542            PrimitiveLiteral::Long(nanos),
543        ) => temporal_scalar(file_dtype, *nanos, TimeUnit::Nanoseconds),
544        (ty, _) => Err(Error::new(
545            ErrorKind::FeatureUnsupported,
546            format!("Predicate literals of type {ty} are not yet supported for vortex data files"),
547        )),
548    }
549}

551/// Builds a literal scalar for comparing against a temporal column.
552///
553/// When the file stores the column as a vortex temporal extension type, the
554/// literal is converted to the column's exact extension dtype (unit and
555/// timezone), since vortex only compares extension values against matching
556/// extension dtypes. Plain primitive columns are compared against the raw
557/// iceberg value.
558fn temporal_scalar(file_dtype: &DType, value: i64, value_unit: TimeUnit) -> Result<Scalar> {
559    match file_dtype {
560        DType::Extension(ext) => {
561            let Some(metadata) = AnyTemporal::try_match(ext) else {
562                return Err(Error::new(
563                    ErrorKind::FeatureUnsupported,
564                    format!(
565                        "Cannot compare a temporal predicate literal to a column of type {file_dtype}"
566                    ),
567                ));
568            };
569            let converted = convert_temporal_value(value, value_unit, metadata.time_unit())?;
570            let nullability = ext.storage_dtype().nullability();
571            let storage = match ext.storage_dtype() {
572                DType::Primitive(PType::I32, _) => Scalar::primitive(
573                    i32::try_from(converted).map_err(|err| {
574                        Error::new(
575                            ErrorKind::DataInvalid,
576                            format!("Temporal value {converted} overflows the column storage type"),
577                        )
578                        .with_source(err)
579                    })?,
580                    nullability,
581                ),
582                DType::Primitive(PType::I64, _) => Scalar::primitive(converted, nullability),
583                dtype => {
584                    return Err(Error::new(
585                        ErrorKind::FeatureUnsupported,
586                        format!("Unsupported temporal storage type {dtype}"),
587                    ));
588                }
589            };
590            Ok(Scalar::extension_ref(ext.clone(), storage))
591        }
592        DType::Primitive(PType::I32, _) => Ok(Scalar::primitive(
593            i32::try_from(value).map_err(|err| {
594                Error::new(
595                    ErrorKind::DataInvalid,
596                    format!("Temporal value {value} overflows the column storage type"),
597                )
598                .with_source(err)
599            })?,
600            Nullability::NonNullable,
601        )),
602        DType::Primitive(PType::I64, _) => Ok(Scalar::primitive(value, Nullability::NonNullable)),
603        dtype => Err(Error::new(
604            ErrorKind::FeatureUnsupported,
605            format!("Cannot compare a temporal predicate literal to a column of type {dtype}"),
606        )),
607    }
608}

711    fn test_batch(ids: Vec<i64>, names: Vec<Option<&str>>) -> RecordBatch {
712        let scores = ids
713            .iter()
714            .map(|id| match id % 3 {
715                0 => None,
716                1 => Some(*id as f64 * 1.5),
717                _ => Some(f64::NAN),
718            })
719            .collect::<Vec<_>>();
720        let timestamps = ids
721            .iter()
722            .map(|id| Some(id * 1_000_000))
723            .collect::<Vec<_>>();
724        let dates = ids
725            .iter()
726            .map(|id| Some(*id as i32 * 10))
727            .collect::<Vec<_>>();
728        let points = Int64Array::from(ids.iter().map(|id| Some(id * 100)).collect::<Vec<_>>());
729        let info = StructArray::new(info_struct_fields(), vec![Arc::new(points)], None);
730        let prices = Decimal128Array::from(
731            ids.iter()
732                .map(|id| Some(i128::from(*id) * 250))
733                .collect::<Vec<_>>(),
734        )
735        .with_precision_and_scale(9, 2)
736        .unwrap();
737        RecordBatch::try_new(test_arrow_schema(), vec![
738            Arc::new(Int64Array::from(ids)),
739            Arc::new(StringArray::from(names)),
740            Arc::new(Float64Array::from(scores)),
741            Arc::new(TimestampMicrosecondArray::from(timestamps)),
742            Arc::new(Date32Array::from(dates)),
743            Arc::new(info),
744            Arc::new(prices),
745        ])
746        .unwrap()
747    }

749    async fn write_test_file(file_path: &str, file_io: &FileIO) -> crate::spec::DataFile {
750        let output = file_io.new_output(file_path).unwrap();
751        let mut writer = VortexWriterBuilder::new(test_schema(), VortexSession::default())
752            .build(output)

766            .pop()

768            .partition_spec_id(0)
769            .build()
770            .unwrap()

787    async fn test_vortex_roundtrip() {
788        let tmp_dir = TempDir::new().unwrap();
789        let file_path = format!("{}/data.vortex", tmp_dir.path().to_str().unwrap());
790        let file_io = FileIO::new_with_fs();
791
792        let data_file = write_test_file(&file_path, &file_io).await;
793        assert_eq!(data_file.file_format(), DataFileFormat::Vortex);
794        assert_eq!(data_file.record_count(), 5);
795        assert_eq!(data_file.value_counts()[&1], 5);
796        assert_eq!(data_file.null_value_counts()[&1], 0);
797        assert_eq!(data_file.null_value_counts()[&2], 1);
798        assert_eq!(data_file.nan_value_counts()[&3], 2);
799
800        // Min/max bounds computed from the written batches. NaN values are
801        // excluded from float bounds; nested fields carry no bounds.
802        assert_eq!(data_file.lower_bounds()[&1], Datum::long(1));
803        assert_eq!(data_file.upper_bounds()[&1], Datum::long(5));
804        assert_eq!(data_file.lower_bounds()[&2], Datum::string("a"));
805        assert_eq!(data_file.upper_bounds()[&2], Datum::string("e"));
806        assert_eq!(data_file.lower_bounds()[&3], Datum::double(1.5));
807        assert_eq!(data_file.upper_bounds()[&3], Datum::double(6.0));
808        assert_eq!(
809            data_file.lower_bounds()[&4],
810            Datum::timestamp_micros(1_000_000)
811        );
812        assert_eq!(
813            data_file.upper_bounds()[&4],
814            Datum::timestamp_micros(5_000_000)
815        );
816        assert_eq!(data_file.lower_bounds()[&5], Datum::date(10));
817        assert_eq!(data_file.upper_bounds()[&5], Datum::date(50));
818        assert_eq!(
819            data_file.lower_bounds()[&8].literal(),
820            &crate::spec::PrimitiveLiteral::Int128(250)
821        );
822        assert_eq!(
823            data_file.upper_bounds()[&8].literal(),
824            &crate::spec::PrimitiveLiteral::Int128(1250)
825        );
826        assert!(!data_file.lower_bounds().contains_key(&6));
827        assert!(!data_file.lower_bounds().contains_key(&7));
828
829        let file_size = std::fs::metadata(&file_path).unwrap().len();
830        assert_eq!(data_file.file_size_in_bytes(), file_size);
831
832        let reader = ArrowReaderBuilder::new(file_io, Runtime::current()).build();
833        let task = test_scan_task(&file_path, test_schema(), file_size);
834        let tasks = Box::pin(futures::stream::iter(vec![Ok(task)])) as FileScanTaskStream;
835        let result = reader.read(tasks).unwrap();
836        let batches: Vec<RecordBatch> = result.stream().try_collect().await.unwrap();
837
838        let total_rows: usize = batches.iter().map(|batch| batch.num_rows()).sum();
839        assert_eq!(total_rows, 5);
840
841        let ids: Vec<i64> = batches
842            .iter()
843            .flat_map(|batch| {
844                batch
845                    .column_by_name("id")
846                    .unwrap()
847                    .as_any()
848                    .downcast_ref::<Int64Array>()
849                    .unwrap()
850                    .values()
851                    .to_vec()
852            })
853            .collect();
854        assert_eq!(ids, vec![1, 2, 3, 4, 5]);
855
856        let names: Vec<Option<String>> = batches
857            .iter()
858            .flat_map(|batch| {
859                batch
860                    .column_by_name("name")
861                    .unwrap()
862                    .as_any()
863                    .downcast_ref::<StringArray>()
864                    .unwrap()
865                    .iter()
866                    .map(|name| name.map(|s| s.to_string()))
867                    .collect::<Vec<_>>()
868            })
869            .collect();
870        assert_eq!(names, vec![
871            Some("a".to_string()),
872            Some("b".to_string()),
873            None,
874            Some("d".to_string()),
875            Some("e".to_string())
876        ]);
877    }
